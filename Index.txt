  <!-- Script Lógico para Cambiar de Compartimento -->
  <script>
    function mostrarSeccion(nombreSeccion) {
      const compartimentos = document.querySelectorAll('.compartimento');
      compartimentos.forEach(comp => comp.classList.remove('activo'));

      const botones = document.querySelectorAll('.nav-btn');
      botones.forEach(btn => btn.classList.remove('active'));

      const seccionObjetivo = document.getElementById(`seccion-${nombreSeccion}`);
      if (seccionObjetivo) {
        seccionObjetivo.classList.add('activo');
      }

      const btnObjetivo = document.getElementById(`btn-${nombreSeccion}`);
      if (btnObjetivo) {
        btnObjetivo.classList.add('active');
      }

      window.scrollTo({ top: 0, behavior: 'smooth' });
    }
  </script>
